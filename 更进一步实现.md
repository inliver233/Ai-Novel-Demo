```python
from fastapi import Response
from typing import AsyncGenerator
import json

class SSEResponse:
    @staticmethod
    async def send_progress(message: str, progress: int) -> str:
        """发送进度更新"""
        data = json.dumps({
            "type": "progress",
            "message": message,
            "progress": progress
        })
        return f"data: {data}\n\n"

    @staticmethod
    async def send_chunk(content: str) -> str:
        """发送内容块"""
        data = json.dumps({
            "type": "chunk",
            "content": content
        })
        return f"data: {data}\n\n"

    @staticmethod
    async def send_result(data: dict) -> str:
        """发送最终结果"""
        result = json.dumps({
            "type": "result",
            "data": data
        })
        return f"data: {result}\n\n"

    @staticmethod
    async def send_error(code: str, message: str) -> str:
        """发送错误"""
        error = json.dumps({
            "type": "error",
            "code": code,
            "message": message
        })
        return f"data: {error}\n\n"
```

```python
from fastapi import APIRouter
from fastapi.responses import StreamingResponse
from app.utils.sse_response import SSEResponse

@router.post("/{chapter_id}/generate")
async def generate_chapter_stream(
    chapter_id: str,
    request: GenerateChapterRequest,
    x_llm_provider: str = Header(...),
    x_llm_api_key: str = Header(...)
) -> StreamingResponse:
    """流式生成章节内容"""

    async def event_generator():
        try:
            # 1. 发送开始进度
            yield await SSEResponse.send_progress("正在准备上下文...", 10)

            # 2. 组装Prompt
            context = await build_chapter_context(chapter_id, request.context)
            yield await SSEResponse.send_progress("正在生成内容...", 30)

            # 3. 流式调用LLM
            llm_client = get_llm_client(x_llm_provider, x_llm_api_key)

            accumulated_content = ""
            async for chunk in llm_client.generate_stream(
                system_prompt=context.system_prompt,
                user_prompt=context.user_prompt
            ):
                accumulated_content += chunk
                yield await SSEResponse.send_chunk(chunk)

            # 4. 解析结果
            yield await SSEResponse.send_progress("正在处理结果...", 90)
            parsed = parse_llm_output(accumulated_content)

            # 5. 发送最终结果
            yield await SSEResponse.send_result({
                "content_md": parsed.content_md,
                "summary": parsed.summary,
                "raw_output": accumulated_content
            })

        except Exception as e:
            yield await SSEResponse.send_error("GENERATION_ERROR", str(e))

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no"  # 禁用Nginx缓冲
        }
    )
```

```typescript
// 文件：frontend/src/services/sseClient.ts
// 参考：MuMuAINovel/frontend/src/utils/sseClient.ts

export class SSEClient {
  private controller: AbortController | null = null;

  async post<T>(
    url: string,
    body: object,
    options: {
      onProgress?: (message: string, progress: number) => void;
      onChunk?: (content: string) => void;
      onResult?: (data: T) => void;
      onError?: (code: string, message: string) => void;
      headers?: Record<string, string>;
    }
  ): Promise<void> {
    this.controller = new AbortController();

    const response = await fetch(url, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        ...options.headers,
      },
      body: JSON.stringify(body),
      signal: this.controller.signal,
    });

    if (!response.ok) {
      throw new Error(`HTTP ${response.status}`);
    }

    const reader = response.body?.getReader();
    if (!reader) throw new Error("No reader available");

    const decoder = new TextDecoder();
    let buffer = "";

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split("\n\n");
      buffer = lines.pop() || "";

      for (const line of lines) {
        if (line.startsWith("data: ")) {
          const data = JSON.parse(line.slice(6));

          switch (data.type) {
            case "progress":
              options.onProgress?.(data.message, data.progress);
              break;
            case "chunk":
              options.onChunk?.(data.content);
              break;
            case "result":
              options.onResult?.(data.data as T);
              break;
            case "error":
              options.onError?.(data.code, data.message);
              break;
          }
        }
      }
    }
  }

  cancel(): void {
    this.controller?.abort();
  }
}
```

#### 记忆管理服务

```python
# 文件：backend/app/services/memory_service.py（新增）
# 参考：Narratium/lib/core/memory-manager.ts (651行)
# 参考：MuMuAINovel/backend/app/services/memory_service.py

from typing import List, Optional
from sentence_transformers import SentenceTransformer
import numpy as np

class MemoryService:
    """记忆管理服务 - 实现四层上下文架构"""

    def __init__(self):
        # 使用多语言embedding模型
        self.embedding_model = SentenceTransformer(
            'paraphrase-multilingual-MiniLM-L12-v2'
        )

    # ========== 第1层：短期对话（最近N轮） ==========
    async def get_recent_context(
        self,
        project_id: str,
        current_chapter_id: str,
        window_size: int = 5  # 最近5章
    ) -> str:
        """获取最近几章的完整内容作为短期上下文"""
        chapters = await self.db.query(Chapter).filter(
            Chapter.project_id == project_id,
            Chapter.number <= current_chapter_number
        ).order_by(Chapter.number.desc()).limit(window_size).all()

        return "\n\n".join([
            f"## 第{ch.number}章 {ch.title}\n{ch.content_md}"
            for ch in reversed(chapters)
        ])

    # ========== 第2层：压缩历史（摘要） ==========
    async def get_compressed_history(
        self,
        project_id: str,
        current_chapter_number: int,
        window_size: int = 5
    ) -> str:
        """获取更早章节的摘要作为压缩历史"""
        chapters = await self.db.query(Chapter).filter(
            Chapter.project_id == project_id,
            Chapter.number < current_chapter_number - window_size
        ).order_by(Chapter.number.asc()).all()

        return "\n".join([
            f"第{ch.number}章 {ch.title}: {ch.summary or '（无摘要）'}"
            for ch in chapters
        ])

    # ========== 第3层：向量记忆库（RAG） ==========
    async def semantic_search(
        self,
        project_id: str,
        query: str,
        top_k: int = 5,
        similarity_threshold: float = 0.7,
        memory_types: Optional[List[MemoryType]] = None
    ) -> List[StoryMemory]:
        """
        语义搜索相关记忆
        参考：Narratium/lib/core/memory-manager.ts 第89-183行
        """
        # 1. 向量化查询
        query_embedding = self.embedding_model.encode(query)

        # 2. 获取该项目所有记忆
        filters = [StoryMemory.project_id == project_id]
        if memory_types:
            filters.append(StoryMemory.memory_type.in_(memory_types))

        memories = await self.db.query(StoryMemory).filter(*filters).all()

        # 3. 计算相似度并排序
        results = []
        for mem in memories:
            if mem.embedding:
                similarity = self._cosine_similarity(
                    query_embedding,
                    np.array(mem.embedding)
                )
                if similarity >= similarity_threshold:
                    # 加权评分：相似度80% + 重要性20%
                    score = similarity * 0.8 + mem.importance_score * 0.2
                    results.append((mem, score))

        # 4. 按分数排序返回Top-K
        results.sort(key=lambda x: x[1], reverse=True)
        return [mem for mem, _ in results[:top_k]]

    async def hybrid_search(
        self,
        project_id: str,
        query: str,
        top_k: int = 5,
        alpha: float = 0.8  # 语义权重
    ) -> List[StoryMemory]:
        """
        混合搜索：语义 + 关键词
        参考：Narratium/lib/core/memory-manager.ts 第188-226行
        """
        # 1. 语义搜索
        semantic_results = await self.semantic_search(
            project_id, query, top_k * 2
        )

        # 2. 关键词搜索
        keywords = self._extract_keywords(query)
        keyword_results = await self.db.query(StoryMemory).filter(
            StoryMemory.project_id == project_id,
            or_(*[StoryMemory.content.contains(kw) for kw in keywords])
        ).limit(top_k * 2).all()

        # 3. 融合结果
        merged = self._merge_results(
            semantic_results, keyword_results, alpha
        )
        return merged[:top_k]

    # ========== 第4层：世界观信息 ==========
    async def get_worldbook_context(
        self,
        project_id: str,
        user_input: str,
        recent_messages: List[str]
    ) -> dict:
        """
        获取匹配的世界观条目
        参考：Narratium/lib/core/world-book.ts 第9-44行
        """
        # 获取项目的WorldBook条目
        entries = await self.get_worldbook_entries(project_id)

        # 构建匹配文本
        match_text = user_input + " ".join(recent_messages[-5:])
        match_text_lower = match_text.lower()

        matched = {"before": [], "after": [], "user_before": [], "user_after": []}

        for entry in entries:
            if not entry.enabled:
                continue

            # 常量条目始终包含
            if entry.constant:
                self._add_to_position(matched, entry)
                continue

            # 关键词匹配
            for key in entry.keys:
                if key.lower() in match_text_lower:
                    self._add_to_position(matched, entry)
                    break

        return matched

    # ========== 自动记忆提取 ==========
    async def extract_memories_from_dialogue(
        self,
        project_id: str,
        chapter_id: str,
        user_message: str,
        assistant_message: str
    ) -> List[dict]:
        """
        使用LLM自动从对话中提取记忆
        参考：Narratium/lib/core/memory-manager.ts 第231-339行
        """
        extraction_prompt = """
分析以下对话，提取值得记住的信息。

对话内容：
用户：{user_message}
AI：{assistant_message}

请识别以下类型的记忆（如果存在）：
1. fact - 事实信息（人名、日期、地点、数字）
2. relationship - 关系动态
3. event - 重要事件
4. preference - 偏好习惯
5. emotion - 情感反应
6. foreshadow - 可能的伏笔

输出JSON格式：
{{
  "memories": [
    {{
      "type": "fact|relationship|event|...",
      "content": "具体内容",
      "importance": 0.0-1.0,
      "tags": ["标签1", "标签2"],
      "confidence": 0.0-1.0
    }}
  ]
}}
"""
        # 调用LLM提取
        result = await self.llm_service.generate(
            extraction_prompt.format(
                user_message=user_message,
                assistant_message=assistant_message
            )
        )

        # 解析并过滤（置信度>=0.6）
        memories = self._parse_extraction_result(result)
        return [m for m in memories if m.get("confidence", 0) >= 0.6]

    # ========== 生成记忆上下文 ==========
    async def generate_memory_context(
        self,
        project_id: str,
        current_input: str,
        max_memories: int = 5
    ) -> str:
        """
        为LLM生成记忆提示词
        参考：Narratium/lib/core/memory-manager.ts 第344-377行
        """
        memories = await self.hybrid_search(
            project_id, current_input, max_memories
        )

        if not memories:
            return ""

        lines = ["相关记忆："]
        for i, mem in enumerate(memories, 1):
            type_label = {
                "fact": "事实",
                "relationship": "关系",
                "event": "事件",
                "preference": "偏好",
                "emotion": "情感",
                "foreshadow": "伏笔"
            }.get(mem.memory_type, "记忆")

            lines.append(f"{i}. [{type_label}] {mem.content}")

        return "\n".join(lines)

    # ========== 工具方法 ==========
    def _cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        """余弦相似度"""
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))
```

```python
            ])

        if context_config.get("include_compressed_history"):

            history = await self.memory_service.get_compressed_history(
```

```python
import httpx
import hashlib
from typing import Dict

class LLMClientPool:
    """LLM HTTP客户端连接池"""

    _pool: Dict[str, httpx.AsyncClient] = {}

    @classmethod
    def get_client(
        cls,
        provider: str,
        base_url: str,
        api_key: str
    ) -> httpx.AsyncClient:
        """
        获取或创建HTTP客户端
        按 provider + base_url + api_key哈希 生成唯一键
        """
        # 生成缓存键
        key_hash = hashlib.md5(
            f"{provider}:{base_url}:{api_key[:8]}".encode()
        ).hexdigest()

        if key_hash not in cls._pool:
            cls._pool[key_hash] = httpx.AsyncClient(
                base_url=base_url,
                headers={
                    "Authorization": f"Bearer {api_key}",
                    "Content-Type": "application/json"
                },
                limits=httpx.Limits(
                    max_keepalive_connections=50,
                    max_connections=100,
                    keepalive_expiry=30.0
                ),
                timeout=httpx.Timeout(
                    connect=60.0,
                    read=180.0,   # LLM可能很慢
                    write=60.0,
                    pool=60.0
                )
            )

        return cls._pool[key_hash]

    @classmethod
    async def close_all(cls):
        """关闭所有连接"""
        for client in cls._pool.values():
            await client.aclose()
        cls._pool.clear()
```

```python
from abc import ABC, abstractmethod
from enum import Enum
from typing import Dict, Any, Optional, List

class NodeCategory(str, Enum):
    ENTRY = "entry"       # 入口节点
    MIDDLE = "middle"     # 中间处理节点
    EXIT = "exit"         # 出口节点
    AFTER = "after"       # 后台异步节点

class NodeStatus(str, Enum):
    NOT_READY = "not_ready"
    READY = "ready"
    RUNNING = "running"
    DONE = "done"
    FAILED = "failed"

class NodeBase(ABC):
    """工作流节点基类"""

    node_name: str = "base"
    category: NodeCategory = NodeCategory.MIDDLE

    def __init__(self, node_id: str, config: Optional[Dict] = None):
        self.node_id = node_id
        self.config = config or {}
        self.status = NodeStatus.NOT_READY
        self.input_data: Dict[str, Any] = {}
        self.output_data: Dict[str, Any] = {}
        self.dependencies: List[str] = []

    @abstractmethod
    async def execute(self, context: "WorkflowContext") -> Dict[str, Any]:
        """执行节点逻辑"""
        pass

    def set_input(self, key: str, value: Any):
        self.input_data[key] = value

    def get_output(self, key: str) -> Any:
        return self.output_data.get(key)
```

```python
# 文件：backend/app/workflow/engine.py
# 参考：Narratium/lib/nodeflow/WorkflowEngine.ts (381行)
# 参考：WriteHERE/recursive/engine.py (617行)

from typing import Dict, List, Optional
import asyncio

class WorkflowEngine:
    """工作流执行引擎"""

    def __init__(self, config: Dict, context: WorkflowContext):
        self.config = config
        self.context = context
        self.nodes: Dict[str, NodeBase] = {}
        self.edges: Dict[str, List[str]] = {}  # 节点依赖关系

    def add_node(self, node: NodeBase):
        self.nodes[node.node_id] = node

    def add_edge(self, from_node: str, to_node: str):
        """添加依赖边：to_node依赖from_node"""
        if to_node not in self.edges:
            self.edges[to_node] = []
        self.edges[to_node].append(from_node)
        self.nodes[to_node].dependencies.append(from_node)

    async def execute(
        self,
        initial_input: Dict[str, Any],
        options: Optional[Dict] = None
    ) -> Dict[str, Any]:
        """
        执行工作流
        参考：Narratium/lib/nodeflow/WorkflowEngine.ts 第103-153行
        """
        options = options or {}

        # 1. 初始化入口节点
        entry_nodes = [
            n for n in self.nodes.values()
            if n.category == NodeCategory.ENTRY
        ]
        for node in entry_nodes:
            node.input_data = initial_input
            node.status = NodeStatus.READY

        # 2. 执行入口节点（并行）
        await asyncio.gather(*[
            self._execute_node(node) for node in entry_nodes
        ])

        # 3. 拓扑排序执行中间节点
        middle_nodes = self._topological_sort()
        for node in middle_nodes:
            if node.category == NodeCategory.EXIT:
                continue
            await self._execute_node(node)

        # 4. 执行出口节点
        exit_nodes = [
            n for n in self.nodes.values()
            if n.category == NodeCategory.EXIT
        ]
        results = await asyncio.gather(*[
            self._execute_node(node) for node in exit_nodes
        ])

        # 5. 后台执行AFTER节点（可选等待）
        after_nodes = [
            n for n in self.nodes.values()
            if n.category == NodeCategory.AFTER
        ]
        if after_nodes:
            if options.get("await_after_nodes", False):
                await asyncio.gather(*[
                    self._execute_node(node) for node in after_nodes
                ])
            else:
                # 不等待，后台执行
                asyncio.create_task(asyncio.gather(*[
                    self._execute_node(node) for node in after_nodes
                ]))

        # 6. 收集结果
        return self._collect_results(exit_nodes)

    async def _execute_node(self, node: NodeBase):
        """执行单个节点"""
        # 收集依赖节点的输出作为输入
        for dep_id in node.dependencies:
            dep_node = self.nodes[dep_id]
            for key, value in dep_node.output_data.items():
                node.set_input(key, value)

        node.status = NodeStatus.RUNNING
        try:
            await node.execute(self.context)
            node.status = NodeStatus.DONE
        except Exception as e:
            node.status = NodeStatus.FAILED
            raise

    def _topological_sort(self) -> List[NodeBase]:
        """拓扑排序（BFS）"""
        # 参考：WriteHERE/recursive/graph.py 第45-77行
        in_degree = {n.node_id: len(n.dependencies) for n in self.nodes.values()}
        queue = [n for n in self.nodes.values() if in_degree[n.node_id] == 0]
        result = []

        while queue:
            node = queue.pop(0)
            if node.category not in (NodeCategory.ENTRY, NodeCategory.AFTER):
                result.append(node)

            for other_id, deps in self.edges.items():
                if node.node_id in deps:
                    in_degree[other_id] -= 1
                    if in_degree[other_id] == 0:
                        queue.append(self.nodes[other_id])

        return result
```

```python
# 文件：backend/app/planning/recursive_planner.py

class TaskType(str, Enum):
    COMPOSITION = "write"   # 写作任务
    REASONING = "think"     # 推理任务
    RETRIEVAL = "search"    # 检索任务

class RecursivePlanner:
    """递归规划器 - 将复杂任务自动分解"""

    async def plan_chapter(
        self,
        chapter_requirement: str,
        max_depth: int = 3
    ) -> TaskGraph:
        """
        将章节生成任务递归分解

        示例输出：
        root (write: 生成第5章)
        ├── think: 分析前4章的核心冲突
        │   └── search: 检索相关角色记忆
        ├── think: 设计本章情节走向
        │   └── think: 规划情感曲线
        ├── write: 撰写开头场景
        │   └── think: 确定开篇钩子
        ├── write: 撰写核心冲突
        │   └── search: 检索相关世界观
        └── write: 撰写结尾悬念
        """
        root = TaskNode(
            task_type=TaskType.COMPOSITION,
            description=chapter_requirement
        )

        await self._recursive_decompose(root, depth=0, max_depth=max_depth)
        return TaskGraph(root)

    async def _recursive_decompose(
        self,
        node: TaskNode,
        depth: int,
        max_depth: int
    ):
        if depth >= max_depth:
            node.is_atomic = True
            return

        # 使用LLM进行任务分解
        sub_tasks = await self._llm_decompose(node)

        for sub in sub_tasks:
            child = TaskNode(**sub)
            node.add_child(child)

            # 递归分解子任务
            if not sub.get("is_atomic"):
                await self._recursive_decompose(child, depth + 1, max_depth)
```

```python
# 参考：MuMuAINovel/backend/app/models/memory.py

class ForeshadowService:
    """伏笔追踪服务"""

    async def plant_foreshadow(
        self,
        project_id: str,
        chapter_id: str,
        content: str,
        strength: float = 0.5
    ) -> StoryMemory:
        """埋下伏笔"""
        return await self.memory_service.create_memory(
            project_id=project_id,
            chapter_id=chapter_id,
            memory_type=MemoryType.FORESHADOW,
            content=content,
            is_foreshadow=1,  # 已埋下
            foreshadow_strength=strength
        )

    async def resolve_foreshadow(
        self,
        memory_id: str,
        resolved_chapter_id: str
    ):
        """回收伏笔"""
        await self.memory_service.update(
            memory_id,
            is_foreshadow=2,  # 已回收
            foreshadow_resolved_at=resolved_chapter_id
        )

    async def get_unresolved_foreshadows(
        self,
        project_id: str
    ) -> List[StoryMemory]:
        """获取未回收的伏笔列表"""
        return await self.db.query(StoryMemory).filter(
            StoryMemory.project_id == project_id,
            StoryMemory.is_foreshadow == 1
        ).all()

    async def suggest_resolution(
        self,
        project_id: str,
        current_chapter: int
    ) -> List[dict]:
        """AI建议应该回收的伏笔"""
        unresolved = await self.get_unresolved_foreshadows(project_id)

        suggestions = []
        for foreshadow in unresolved:
            # 根据埋下时间和强度计算回收紧迫度
            chapters_since = current_chapter - foreshadow.story_timeline
            urgency = foreshadow.foreshadow_strength * (chapters_since / 10)

            if urgency > 0.5:
                suggestions.append({
                    "memory": foreshadow,
                    "urgency": urgency,
                    "suggestion": f"建议在本章回收伏笔：{foreshadow.content[:50]}..."
                })

        return sorted(suggestions, key=lambda x: x["urgency"], reverse=True)
```

## 6. 技术选型推荐

### 6.1 向量数据库选择

| 方案 | 优点 | 缺点 | 推荐场景 |
|-----|------|------|---------|
| **PostgreSQL + pgvector** | 统一数据库、简单部署 | 性能一般 | MVP、中小规模 |
| ChromaDB | 易用、开箱即用 | 单机、难扩展 | 本地开发 |
| Pinecone | 云托管、高性能 | 收费、依赖网络 | 生产环境 |
| Weaviate | 开源、功能丰富 | 部署复杂 | 企业级 |

**推荐**：Phase 2使用pgvector，Phase 4+考虑Pinecone

### 6.2 Embedding模型选择

| 模型 | 维度 | 多语言 | 成本 | 推荐 |
|-----|------|-------|------|------|
| text-embedding-3-small (OpenAI) | 1536 | 是 | $0.02/1M | Phase 2+ |
| paraphrase-multilingual-MiniLM-L12-v2 | 384 | 是 | 免费本地 | MVP |
| m3e-base (中文优化) | 768 | 中文 | 免费本地 | 中文为主 |

### 6.3 任务队列选择

| 方案 | 优点 | 缺点 | 推荐 |
|-----|------|------|------|
| Celery + Redis | 成熟、生态丰富 | 配置复杂 | 生产环境 |
| RQ (Redis Queue) | 简单易用 | 功能有限 | 小规模 |
| 内存队列 (asyncio.Queue) | 零依赖 | 不持久化 | MVP |

1. **Phase 2必做**：
   - SSE流式输出（用户体验大幅提升）
   - 多层上下文管理（解决长篇小说记忆问题）
   - HTTP客户端池优化（性能提升）

2. **Phase 3推荐**：
   - 工作流引擎（架构升级，便于扩展）
   - 向量记忆系统（RAG能力）
   - 章节分析功能（用户粘性）

3. **Phase 4可选**：
   - 递归规划引擎（复杂任务自动分解）
   - 知识图谱（高级一致性保证）
   - 伏笔追踪（专业写作辅助）

4. **Phase 5长期**：
   - 插件系统（生态建设）
   - 分布式任务队列（高可用）
   - 多用户协作（平台化）

### 最有价值的借鉴

1. **Narratium**：四层上下文架构 + NodeFlow工作流
2. **MuMuAINovel**：SSE流式 + 章节分析 + 生产级部署
3. **WriteHERE**：递归规划 + 任务自动分解
4. **马良AI**：多模型支持 + 内容提供器工厂
5. **NovelForge**：工作流引擎 + 卡片系统
